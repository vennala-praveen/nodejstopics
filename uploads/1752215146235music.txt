import React, { useState } from 'react';
import './MusicPlayer.css'
import AudioPlayer from 'react-h5-audio-player';
import 'react-h5-audio-player/lib/styles.css';

import Tauba from './assets/songs/Tauba.mp3';
import KaiseHua from './assets/songs/KaiseHua.mp3';
import TaubaImg from './assets/images/tauba.jpg';
import KaiseHuaImg from './assets/images/kaisehua.jpg';



const MusicPlayer = () => {

  const [currentIndex, setIndex] = useState(0);

  const tracks = [
    {
      songName: 'Tauba Tauba',
      song: Tauba,
      songImage: TaubaImg
    },
    {
      songName: 'Kaise Hua',
      song: KaiseHua,
      songImage: KaiseHuaImg
    }

  ];

  const handleClickNext = () => {
    setIndex((currentIndex + 1) % tracks.length);
  };

  const handleClickPrevious = () => {
    setIndex((currentIndex - 1 + tracks.length) % tracks.length);
  }

  return (
    <>
      <div className='music-player-card'>
        <img className='track-image' src={tracks[currentIndex].songImage} alt='songImage' />
        <h2> {tracks[currentIndex].songName}</h2>
        <AudioPlayer
          src={tracks[currentIndex].song}
          onPlay={e => console.log("onPlay")}
          onClickNext={() => { handleClickNext() }}
          onClickPrevious={() => { handleClickPrevious() }}
          showSkipControls={true}
          showJumpControls={false}
        />
      </div>
    </>
  );
};

export default MusicPlayer;
